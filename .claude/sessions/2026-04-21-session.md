33	def _resolve_schema_from_path(path: str) -> type[BaseModel]:
34	    """Resolve a dotted path like 'a.b.Class' to the BaseModel subclass.
35	
36	    The skill loader registers dynamically loaded modules in sys.modules under
37	    a namespaced key (``_graph_agent_skill_.<hash>.<module>``); classes defined
38	    in those modules carry the namespaced ``__module__``. Looking up via
39	    ``sys.modules`` first avoids importlib re-import attempts that fail on
40	    namespaced dotted paths.
41	    """
42	    module_str, _, cls_name = path.rpartition(".")
43	    if not module_str or not cls_name:
44	        raise ValueError(f"invalid schema path: {path!r}")
45	    module = sys.modules.get(module_str)
46	    if module is None:
47	        module = importlib.import_module(module_str)
48	    cls = getattr(module, cls_name, None)
49	    if cls is None:
50	        raise ValueError(f"schema path {path!r} resolved module has no attribute {cls_name!r}")
51	    if not (isinstance(cls, type) and issubclass(cls, BaseModel)):
52	        got = type(cls).__name__
53	        raise ValueError(f"schema path {path!r} is not a BaseModel subclass (got {got})")
54	    return cls

158	def _unwrap_optional(annotation: Any) -> Any:
159	    """For T | None or Optional[T], return T. Otherwise return annotation."""
160	    # typing.Optional[T] is typing.Union[T, None]
161	    if typing.get_origin(annotation) is typing.Union:
162	        args = [a for a in typing.get_args(annotation) if a is not type(None)]
163	        if len(args) == 1:
164	            return args[0]
165	    # Python 3.10+ union syntax: X | Y
166	    if isinstance(annotation, types.UnionType):
167	        args = [a for a in annotation.__args__ if a is not type(None)]  # type: ignore[union-attr]
168	        if len(args) == 1:

170	    return annotation

202	def _parse_at_key_lines(raw_lines: list[str]) -> list[dict[str, str]]:
203	    """Parse ``@key: val`` indented lines into a list of dicts.
204	
205	    A repeated key signals the start of a new sub-object.
206	    Non-@key lines are skipped (they should not appear in an @key block; a warning is logged).
207	
208	    Example input:
209	        ["@speaker: 旁白", "@text: 她回头", "@speaker: 主角", "@text: 来了"]
210	
211	    Example output:
212	        [{"speaker": "旁白", "text": "她回头"}, {"speaker": "主角", "text": "来了"}]
213	    """
214	    objects: list[dict[str, str]] = []
215	    current: dict[str, str] = {}
216	    for raw in raw_lines:
217	        m = re.match(r"@(\w+):\s*(.*)", raw.strip())
218	        if m:
219	            key, val = m.group(1), m.group(2).strip()
220	            if key in current:
221	                # Repeated key → close current object, start new one
222	                objects.append(current)
223	                current = {}
224	            current[key] = val
225	        else:
226	            stripped = raw.strip()
227	            if stripped:
228	                logger.warning(
229	                    "parse_md: non-@key line in nested sub-object context, skipping: %r",
230	                    stripped,
231	                )
232	    if current:
233	        objects.append(current)
234	    return objects

254	    """Parse structured Markdown text into raw item dicts.

259

261	    prefix that guarantees no collision with schema field names.

264	    """
265	    annotations = _get_field_annotations(schema)
266	    blocks = _split_into_blocks(md_text)

268	
269	    items: list[dict[str, Any]] = []
270	    for item_id, block_lines in blocks:
271	        item = _parse_block(item_id, block_lines, annotations)
272	        items.append(item)
273	
274	    logger.info("parse_md: schema=%s parsed=%d items", schema.__name__, len(items))
275	    return items

278	def _split_into_blocks(md_text: str) -> list[tuple[str, list[str]]]:
279	    """Split MD text on ``## `` markers → [(item_id, body_lines), ...]."""
280	    blocks: list[tuple[str, list[str]]] = []
281	    current_id: str | None = None
282	    current_lines: list[str] = []

284	    for raw_line in md_text.splitlines():

289	            current_id = m.group(1).strip()
290	            current_lines = []
291	        elif current_id is not None:
292	            current_lines.append(raw_line)
293	
294	    if current_id is not None:
295	        blocks.append((current_id, current_lines))

297	    return blocks

300	def _parse_block(
301	    item_id: str,
302	    lines: list[str],
303	    annotations: dict[str, Any],
304	) -> dict[str, Any]:
305	    """Parse the body lines of one ## block into a flat field dict."""
306	    item: dict[str, Any] = {"_md_id": item_id}
307	    current_nested_key: str | None = None
308	    nested_children: list[str] = []
309	
310	    def _flush_nested() -> None:
311	        """Apply accumulated children to current_nested_key."""
312	        nonlocal current_nested_key, nested_children
313	        if current_nested_key is None:
314	            return
315	        ann = annotations.get(current_nested_key)
316	        if _is_list_annotation(ann):
317	            inner_type = _get_list_inner_type(ann)
318	            # Check if inner_type is a BaseModel subclass → @key format
319	            if isinstance(inner_type, type) and issubclass(inner_type, BaseModel):
320	                item[current_nested_key] = _parse_at_key_lines(nested_children)
321	            elif any(c.strip().startswith("@") for c in nested_children):
322	                # Annotation says list[str] but children look like @key lines →
323	                # still parse as sub-objects (LLM sometimes uses @key for dict lists)
324	                item[current_nested_key] = _parse_at_key_lines(nested_children)
325	            else:
326	                # Plain list[str]
327	                item[current_nested_key] = [c.strip() for c in nested_children if c.strip()]
328	        else:
329	            # Non-list nested field — join children (unlikely, but handle gracefully)
330	            item[current_nested_key] = ", ".join(c.strip() for c in nested_children if c.strip())
331	        current_nested_key = None
332	        nested_children = []
333	
334	    for line in lines:
335	        if not line.strip():
336	            continue  # skip blank lines
337	
338	        # Indented child — must be checked FIRST (before flat/nested patterns)
339	        child_m = _RE_INDENTED_CHILD.match(line)
340	        if child_m:
341	            if current_nested_key is not None:
342	                nested_children.append(child_m.group(1))
343	            else:
344	                logger.warning(
345	                    "parse_md: indented child outside nested field, skipping: %r",
346	                    line,
347	                )
348	            continue
349	
350	        # New top-level bullet → flush any pending nested field
351	        _flush_nested()
352	
353	        # Flat field: "- key: value"
354	        flat_m = _RE_FLAT_FIELD.match(line)
355	        if flat_m:
356	            key, raw_val = flat_m.group(1), flat_m.group(2).strip()
357	            item[key] = _coerce_scalar(key, raw_val, annotations)
358	            continue
359	
360	        # Nested field start: "- key:"
361	        nested_m = _RE_NESTED_FIELD.match(line)
362	        if nested_m:
363	            current_nested_key = nested_m.group(1)
364	            nested_children = []
365	            continue
366	
367	        logger.warning("parse_md: unrecognised line, skipping: %r", line)
368	
369	    # Flush the last pending nested field
370	    _flush_nested()
371	    return item

411	    items: list[dict[str, Any]], schema: type[T]
412	) -> DiagnosticReport:
413	    """Validate each item dict against ``schema`` independently.
414	
415	    One item failing validation does NOT affect any other item.
416	    Valid items are collected in ``DiagnosticReport.valid_items``;
417	    failures become ``DiagnosticReport.errors`` entries with per-field error details.
418	    """
419	    valid_items: list[BaseModel] = []
420	    errors: list[ItemError] = []
421	
422	    for i, item in enumerate(items):
423	        try:
424	            valid_items.append(schema.model_validate(item))
425	        except PydanticValidationError as exc:
426	            field_errors = [

429	                    error=err["msg"],

432	                for err in exc.errors()

436	                    index=i,
437	                    item_id=item.get("_md_id"),  # type: ignore[arg-type]
438	                    fields=field_errors,

440	            )
441	
442	    logger.info(

444	        schema.__name__,

447	        len(errors),

454	def _extract_md_excerpt(md_text: str, error_indices: set[int]) -> str:
455	    """Extract only the ## blocks at ``error_indices`` from ``md_text``.
456	
457	    Implementation:
458	      1. Split md_text on lines that start a new ## block (keeping the delimiter).
459	      2. Filter to only item blocks (skip any pre-header preamble).
460	      3. Select the sub-list at error_indices.
461	      4. Rejoin for the Patch Agent prompt.
462	    """
463	    # Split on lines that start a new ## block; re.split with lookahead keeps delimiters
464	    raw_parts = re.split(r"(?m)^(?=##\s)", md_text)
465	    # Keep only parts that begin with '## ' (i.e. actual item blocks).
466	    # Use simple prefix check because p is a multi-line chunk, not a single line.
467	    item_parts = [p for p in raw_parts if p.lstrip().startswith("## ")]
468	    selected = [item_parts[i] for i in sorted(error_indices) if i < len(item_parts)]
469	    return "\n".join(selected)

573	    import typing

577	    import types

581	    # Check for Optional/Union types (typing.Union or types.UnionType for X | Y syntax)
582	    is_union = False

584	        is_union = True

617	            for m in field_info.metadata:
618	                if hasattr(m, 'ge'):
619	                    ge_val = m.ge
620	                elif m.__class__.__name__ == 'Ge' and hasattr(m, 'ge'):
621	                    ge_val = m.ge
622	                if hasattr(m, 'le'):
623	                    le_val = m.le
624	                elif m.__class__.__name__ == 'Le' and hasattr(m, 'le'):
625	                    le_val = m.le
626	        if ge_val is not None:
627	            constraint += f", >={ge_val}"
628	        if le_val is not None:
629	            constraint += f", <={le_val}"
630	        constraint += "]"
631	        return constraint
632	
633	    if annotation is float:
634	        constraint = "[小数"
635	        ge_val = None
636	        le_val = None
637	        if field_info and hasattr(field_info, 'metadata'):
638	            for m in field_info.metadata:
639	                if hasattr(m, 'ge'):
640	                    ge_val = m.ge
641	                elif m.__class__.__name__ == 'Ge' and hasattr(m, 'ge'):

## graph_agent 全面分析报告（IMG_0076–IMG_0085）

**位置**：src/core/graph_agent/（内嵌 DeerFlow），~142 个 Python 文件，~4000 行核心代码。  
**定位**：以 SKILL.md 为单一事实源的声明式多阶段 Agent 编排引擎。

### 一、功能能力

#### 1.1 核心抽象（public API）

graph_agent/__init__.py 导出的对外原语：

| 原语 | 类型 | 职责 |
|------|------|------|
| run_skill | function | 一键执行 SKILL.md，无需手写 harness |
| GraphAgentHarness | class | 外层编排器（LangGraph 驱动）|
| Phase | dataclass | 单阶段配置（prompt/tools/validator/…）|
| ContextBridge | dataclass | 父子 skill 输入输出映射 |
| WorkflowState | TypedDict | 节点点流转状态（context/messages/…）|
| load_workflow_from_md | function | SKILL.md → GraphAgentHarness |
| compile_skill | function | 静态规则校验（rules.yaml）|

| ModelResolver | class | role → model → provider 解析 |

#### 1.2 五类核心能力

1. **声明式多阶段工作流**：SKILL.md 里的 `<node>/<phase_config>` 编译成 list[Phase]，按照顺序在 StateGraph 中流转。

2. **认知循环约束（cognitive/）**：强制 LLM 写 working_memory（本阶段目标/步骤/数据来源），用 finish_task() 结构化自检（reasoning/evidence/plan_checklist），并通过 DeadEndPruningMiddleware 识别死循环。

3. **子技能调用（Sub-skill）**：core/skill_tool_factory.py，SubSkillSpec 把任意 SKILL.md 包成 StructuredTool，零 Python 跨 skill 互调。

4. **MD→Pydantic 自动解析修复**：tools/md_to_json.py + skills/builtin/md-patch/——LLM 输出被解析成 Pydantic；失败时用 "md-patch" 小模型做 surgical 修复，避免整轮重跑。

5. **角色化模型路由**：models/resolver.py——tier → role → model_codes[] → provider_codes[]，任一层失败自动降级；30 min 熔断窗口。

### 二、工程架构

#### 2.1 双层控制架构（核心设计点）

```
外层 GraphAgentHarness（LangGraph）
  • 阶段顺序与条件路由（phase→validator→retry→next）
  • planning/selfcheck nudge 注入
  • checkpoint compaction（长任务消息压缩）
  • validator 重试路由到 retry_target

          per-phase .invoke(state)
                    ↓
  内层 DeerFlow create_agent() loop
  • LLM 调用 + tool 执行 + 流式响应
  • 中间件：WorkingMemory / DeadEndPruning / ToolError
  • Sandbox 工具（list_dir/read_file/execute_command）
  • SubagentExecutor(max_workers=3) 用于并行子任务
```

**设计原则**（docs/graph_agent_docs/ARCHITECTURE.md）：
- 不改 DeerFlow 源码——通过外层 harness、callbacks、middleware、config 解决问题
- 框架层零业务逻辑——业务只写在 skill 目录里
- Kitchen-Pass（厨房传递）：一阶产物结果先进 context，真正持久化由 IOManager + 调用方注入的 artifact_saver 完成

|------|------|------|
| core/ | ~3500 | Harness/Runner/Loader/Compiler/Parser/State |
| cognitive/ | ~560 | prompt/finish/memory/ambiguity/middlewares |

| config/ | ~880 | llm_roles.yaml / multimodal_roles.yaml 等 |

### 一、Claude Code SDK vs 子进程科普

两者都是"让自己的程序用上 Claude Code 的能力"，但路径不同。

#### 子进程方式（CLI subprocess）

原理：你的程序 spawn `claude` 命令作为子进程，通过 stdin/stdout 通信。

```python
your_server.py
  └── subprocess.Popen(["claude", "-p", task, "--output-format", ...])
      → 把 PM 对话写入 stdin
      → 解析子进程输出的 JSON 事件（tool_use / text / result）
```

✅ 继承用户本地所有配置：~/.claude/CLAUDE.md、MCP、自定义 skills、settings.json 全部生效  
✅ 实现超简单（几十行）  
❌ 部署困难：用户必须安装 Claude Code CLI 才能用 Studio  
❌ 程序化拦截 tool_use（做 diff 预览）需要解析文本流，容易漂移

#### SDK 方式（Claude Agent SDK）

原理：Python/TS 的 SDK 库，内部是一个完整的 agent loop 实现，把 Claude Code 的核心能力封装成函数/类。

```python
from claude_agent_sdk import query, ClaudeAgentOptions

async for msg in query(
    prompt="给 SKILL.md 加一个事件提取阶段",
    options=ClaudeAgentOptions(
        allowed_tools=["Read", "Edit", "Glob"],  # 代码精细控制
        cwd="/path/to/skill/dir",
        mcp_servers={...},
        system_prompt_suffix="...",    # 追加到系统提示
    )
):
    # msg 是结构化对象：ToolUseBlock、TextBlock、ResultBlock
    if isinstance(msg, ToolUseBlock) and msg.name == "Edit":
        # 拦截并展示 diff 预览
```

✅ 程序化控制强：tool_use 是类型对象，拦截/审批/改写都容易  
✅ 部署轻量：pip install claude-agent-sdk，用户不需要装 CLI  
✅ Studio 可以给每个 PM 会话独立的权限策略  
❌ 不自动继承用户本地的 CLAUDE.md / skills——要在代码里显式加载  
⚠️ SDK 不等于"完整的 Claude Code"：不带交互式 UI、不带 slash commands、不带 IDE 集成；但核心 agent loop + 工具 + sub-agent + MCP + hooks 都有

#### SDK 是把 Claude Code 完整接进来吗？

核心引擎接进来了，外壳没接进来。具体说：

| 能力 | CLI 有 | SDK 有 |
|------|--------|--------|
| Agent loop（思考→工具→响应）| ✓ | ✓ |
| 工具：Read/Edit/Write/Bash/Glob/Grep | ✓ | ✓ |
| Sub-agent（Task 工具）| ✓ | ✓ |
| MCP servers | ✓ | ✓ |
| Hooks | ✓ | ✓（更细粒度）|
| 权限模式 | ✓ | ✓（更细粒度）|
| 自定义 Skills（Agent Skills）| ✓ | ✓（需要 setting…）|
| 交互式 TUI | ✓ | ✗（SDK 是库，U…）|
| /init/clear/review 等 slash commands | ✓ | ✗（但可用 setti…）|

**对 Skill Studio 的选型结论**：Studio 用 SDK。理由：web 服务化部署 + 需要程序化拦截工具调用做 diff 预览 + 每个 PM 独立权限沙箱。子进程在作为开发者 fallback。

### 二、工程架构

#### 2.1 双层控制架构（核心设计点）

```
外层 GraphAgentHarness（LangGraph）
  • 阶段顺序与条件路由（phase→validator→retry→next）
  • planning/selfcheck nudge 注入
  • checkpoint compaction（长任务消息压缩）
  • validator 重试路由到 retry_target

          per-phase .invoke(state)
                    ↓
  内层 DeerFlow create_agent() loop
  • LLM 调用 + tool 执行 + 流式响应
  • 中间件：WorkingMemory / DeadEndPruning / ToolError
  • Sandbox 工具（list_dir/read_file/execute_command）
  • SubagentExecutor(max_workers=3) 用于并行子任务
```

**设计原则**（docs/graph_agent_docs/ARCHITECTURE.md）：
- 不改 DeerFlow 源码——通过外层 harness、callbacks、middleware、config 解决问题
- 框架层零业务逻辑——业务只写在 skill 目录里
- Kitchen-Pass（厨房传递）：一阶产物结果先进 context，真正持久化由 IOManager + 调用方注入的 artifact_saver 完成

### 一、Claude Code SDK vs 子进程科普

两者都是"让自己的程序用上 Claude Code 的能力"，但路径不同。

#### 子进程方式（CLI subprocess）

原理：你的程序 spawn `claude` 命令作为子进程，通过 stdin/stdout 通信。

```python
your_server.py
  └── subprocess.Popen(["claude", "-p", task, "--output-format", ...])
      → 把 PM 对话写入 stdin
      → 解析子进程输出的 JSON 事件（tool_use / text / result）
```

✅ 继承用户本地所有配置：~/.claude/CLAUDE.md、MCP、自定义 skills、settings.json 全部生效  
✅ 实现超简单（几十行）  
❌ 部署困难：用户必须安装 Claude Code CLI 才能用 Studio  
❌ 程序化拦截 tool_use（做 diff 预览）需要解析文本流，容易漂移

#### SDK 方式（Claude Agent SDK）

原理：Python/TS 的 SDK 库，内部是一个完整的 agent loop 实现，把 Claude Code 的核心能力封装成函数/类。

```python
from claude_agent_sdk import query, ClaudeAgentOptions

async for msg in query(
    prompt="给 SKILL.md 加一个事件提取阶段",
    options=ClaudeAgentOptions(
        allowed_tools=["Read", "Edit", "Glob"],  # 代码精细控制
        cwd="/path/to/skill/dir",
        mcp_servers={...},
        system_prompt_suffix="...",    # 追加到系统提示
    )
):
    # msg 是结构化对象：ToolUseBlock、TextBlock、ResultBlock
    if isinstance(msg, ToolUseBlock) and msg.name == "Edit":
        # 拦截并展示 diff 预览
```

✅ 程序化控制强：tool_use 是类型对象，拦截/审批/改写都容易  
✅ 部署轻量：pip install claude-agent-sdk，用户不需要装 CLI  
✅ Studio 可以给每个 PM 会话独立的权限策略  
❌ 不自动继承用户本地的 CLAUDE.md / skills——要在代码里显式加载  
⚠️ SDK 不等于"完整的 Claude Code"：不带交互式 UI、不带 slash commands、不带 IDE 集成；但核心 agent loop + 工具 + sub-agent + MCP + hooks 都有

#### SDK 是把 Claude Code 完整接进来吗？

核心引擎接进来了，外壳没接进来。具体说：

| 能力 | CLI 有 | SDK 有 |
|------|--------|--------|
| Agent loop（思考→工具→响应）| ✓ | ✓ |
| 工具：Read/Edit/Write/Bash/Glob/Grep | ✓ | ✓ |
| Sub-agent（Task 工具）| ✓ | ✓ |
| MCP servers | ✓ | ✓ |
| Hooks | ✓ | ✓（更细粒度）|
| 权限模式 | ✓ | ✓（更细粒度）|
| 自定义 Skills（Agent Skills）| ✓ | ✓（需要 setting…）|
| 交互式 TUI | ✓ | ✗（SDK 是库，U…）|
| /init/clear/review 等 slash commands | ✓ | ✗（但可用 setti…）|

**对 Skill Studio 的选型结论**：Studio 用 SDK。理由：web 服务化部署 + 需要程序化拦截工具调用做 diff 预览 + 每个 PM 独立权限沙箱。子进程在作为开发者 fallback。
